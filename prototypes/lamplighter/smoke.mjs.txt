import { chromium } from '/home/user/margwise-phase2/node_modules/.pnpm/playwright-core@1.64.1/node_modules/playwright-core/index.mjs';
const b = await chromium.launch({ executablePath: '/opt/pw-browsers/chromium' });
const H = Number(process.env.H || 780);
const ctx = await b.newContext({ viewport: { width: 393, height: H }, deviceScaleFactor: 2, isMobile: true, hasTouch: true });
const p = await ctx.newPage();
const errs = []; p.on('pageerror', e => errs.push(String(e))); p.on('console', m => { if (m.type()==='error') errs.push(m.text()); });
await p.goto('file://' + process.cwd() + '/lamplighter.html');
await p.waitForTimeout(800);
await p.screenshot({ path: 's3-welcome.png' });
await p.click('#begin'); await p.waitForTimeout(1500);
await p.click('.opt >> nth=0'); await p.click('.opt >> nth=0'); await p.waitForTimeout(700);
await p.screenshot({ path: 's3-stop1.png' });
// overflow check
const ov = async (tag) => console.log(tag, JSON.stringify(await p.evaluate(() => ({ sh: document.documentElement.scrollHeight, ih: innerHeight, bodyOv: document.body.scrollHeight }))));
await ov('stop1');
let stops = 0, sawEcho = false;
for (let i = 0; i < 20; i++) {
  const visible = await p.evaluate(() => document.querySelector('#stop').classList.contains('on'));
  if (!visible) break;
  const echo = await p.evaluate(() => /flickered/.test(document.querySelector('#bubble').textContent));
  if (echo) sawEcho = true;
  await p.click('.opt >> nth=' + (i % 4)); await p.waitForTimeout(150);
  if (i % 3 === 0) { await p.click('.opt >> nth=' + ((i + 1) % 4)); }
  await p.click('#light'); stops++;
  await p.waitForTimeout(2300);
  if (i === 4) await p.screenshot({ path: 's3-stop6.png' });
}
console.log('stops', stops, 'sawEcho', sawEcho);
await p.screenshot({ path: 's3-dreams.png' }); await ov('dreams');
await p.click('.chip >> nth=0'); await p.click('.chip >> nth=5'); await p.click('#dreamGo'); await p.waitForTimeout(1500);
await p.screenshot({ path: 's3-reveal.png' }); await ov('reveal');
await p.click('#openDoors'); await p.waitForTimeout(800);
await p.screenshot({ path: 's3-doors.png' }); await ov('doors');
console.log('ERRORS', JSON.stringify(errs));
await b.close();
